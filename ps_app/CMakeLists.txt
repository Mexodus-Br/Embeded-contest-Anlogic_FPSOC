cmake_minimum_required(VERSION 3.16)
project(ps_d1_handoff LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

set(PS_KIT "${CMAKE_SOURCE_DIR}/ps_partner_v1_1_20261022_005289b/ps_partner_v1")
add_library(ps_core src/ps_core.cpp "${PS_KIT}/golden_model/src/model.cpp"
  "${PS_KIT}/pl_rs485/software/rs485_gp.c" "${PS_KIT}/pl_rs485/software/rs485_telem.c")
target_include_directories(ps_core PUBLIC src third_party "${PS_KIT}")
add_executable(ps_core_cli src/ps_cli.cpp)
set_target_properties(ps_core_cli PROPERTIES OUTPUT_NAME ps_core)
target_link_libraries(ps_core_cli PRIVATE ps_core)
find_package(Threads REQUIRED)
add_executable(ps_server src/ps_backend.cpp src/ps_runtime.cpp src/ps_service.cpp src/ps_server.cpp)
target_link_libraries(ps_server PRIVATE ps_core Threads::Threads)
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND CMAKE_CXX_COMPILER_VERSION VERSION_LESS 9)
  target_link_libraries(ps_server PRIVATE stdc++fs)
endif()
if(WIN32)
  target_link_libraries(ps_server PRIVATE ws2_32 bcrypt)
endif()
foreach(target ps_core ps_core_cli ps_server)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /utf-8 /Od /fp:strict)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -O0 -fno-fast-math -ffp-contract=off)
  endif()
endforeach()
